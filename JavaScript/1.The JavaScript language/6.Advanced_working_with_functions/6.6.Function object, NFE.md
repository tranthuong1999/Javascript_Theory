1. Functions are objects
1.1. Function trong JavaScript là gì?
Function có thể được lưu vào biến
<!-- 
function sayHi() {
    console.log("Hello");
}

let anotherFunction = sayHi;

anotherFunction(); // Hello
 -->
Nhưng có 1 điểm quan trọng cần hiểu :
Trong JS , function cũng là 1 object
Điều này có nghĩa là function không chỉ được gọi bằng () , mà còn có thể :
1. Gán cho biến.
2. Truyền vào function khác
3. Trả về từ function khác
4. Có các thuộc tính 
5. Có các phương thức

Ví dụ:
<!-- 
function sayHi() {
    console.log("Hello");
}

// Thêm property cho function
sayHi.description = "A greeting function";

// Truy cập property
console.log(sayHi.description);

// Gọi function
sayHi();
 -->
1.2. Hình dung Function như một object
sayHi (Function object)
│
├── Có thể gọi
│   └── sayHi()
│
├── Properties
│   └── description = "A greeting function"
│
└── Built-in properties
    ├── name
    └── length
Function vừa có khả năng thực thi code vùa có khả năng lưu trữ dữ liệu như 1 object
1.3. Kiểm tra kiến thức
<!-- 
function sayHi() {
    console.log("Hello");
}

sayHi.age = 27;

console.log(sayHi.age);
console.log(typeof sayHi);
console.log(typeof sayHi.age);
 -->
Giải thích:
- sayHi.age = 27: thêm property age vào function object.
- typeof sayHi: trả về "function" vì sayHi là function.
- typeof sayHi.age: trả về "number" vì giá trị của property age là số 27.
Điểm cần nhớ:Function là 1 object đặc biệt có thể gọi được (allable object) . VÌ vậy nó vừa có thể thực thi code ,
vừa có thể lưu trữ properties.

2. The "name" property
2.1. Thuộc tính name
Mỗi function thường có 1 proprty tên name , dùng để lấy tên function
<!-- 
function sayHi() {
    console.log("Hello");
}

console.log(sayHi.name); // sayHi
 -->
JS tự động xác định tên function trong nhiều trường hợp
Ví dụ:
<!-- 
function sayHi() {}

let sayBye = function() {};

let sayHello = () => {};

console.log(sayHi.name);    // sayHi
console.log(sayBye.name);   // sayBye
console.log(sayHello.name); // sayHello
 -->
Ngay cả khi function không được viết tên , JS vẫn có thể suy luận từ nơi nó được gán
Đây được gọi là contextual name.
2.2. Function trong object
<!-- 
let user = {
    sayHi() {
        console.log("Hello");
    },

    sayBye: function() {
        console.log("Bye");
    }
};

console.log(user.sayHi.name);  // sayHi
console.log(user.sayBye.name); // sayBye
 -->
2.3. Trường hợp không xác định được tên
<!-- 
let arr = [function() {}];
console.log(arr[0].name); // ""
 -->
 Bài tập nhỏ :
 <!-- 
 let user = {
    firstName: "Martin",

    sayHi: function() {
        console.log("Hello");
    }
};

let fn = user.sayHi;

console.log(user.sayHi.name);
console.log(fn.name);
  -->

3. The "length" property
3.1. Thuộc tính length là gì?
Function có thuộc tính length , dùng để lấy số lượng parameters được khai báo trong function
<!-- 
function sum(a, b) {
    return a + b;
}

console.log(sum.length); // 2
 -->
3.2. Một số trường hợp đặc biệt
Trường hợp 1: Default parameters
<!-- 
function show(a, b, c = 10) {}

console.log(show.length); // 2
 -->
 Chỉ đếm các parameters trước parameter đầu tiên có giá trị mặc định

Trường hợp 2: Rest parameters
<!-- 
function show(a, b, ...args) {}

console.log(show.length); // 2
 -->
Rest parameter k được tính

Trường hợp 3: Tất cả đều có default value
<!-- 
function show(a = 1, b = 2) {}

console.log(show.length); // 0
 -->
3.3. Bảng ghi nhớ
| Function                     | `length` |
| ---------------------------- | -------: |
| `function fn(a, b)`          |        2 |
| `function fn(a, b, c = 10)`  |        2 |
| `function fn(a, b, ...args)` |        2 |
| `function fn(a = 1, b = 2)`  |        0 |
<!-- 

function test(a, b, c = 10, d, ...rest) {}
console.log(test.length);
 -->
| Parameter | Được tính? | Lý do                          |
| --------- | ---------- | ------------------------------ |
| `a`       | Có         | Trước default parameter        |
| `b`       | Có         | Trước default parameter        |
| `c = 10`  | Không      | Có default value               |
| `d`       | Không      | Đứng sau default parameter     |
| `...rest` | Không      | Rest parameter không được tính |

3.4. Ứng dụng trong callback handlers
Thuộc tính length có thể được sử dụng để kiểm tra số lượng parameters mà 1 function nhận vào
<!-- 
function ask(question, ...handlers) {
    let isConfirm = confirm(question);

    for (let handler of handlers) {
        if (handler.length === 0) {
            if (isConfirm) handler();
        } else {
            handler(isConfirm);
        }
    }
}

ask(
    "Do you agree?",
    () => console.log("You agreed"),
    result => console.log(result)
);
 -->
Ở đây:
- Callback () => ... có length = 0.
- Callback result => ... có length = 1.
- Function ask dựa vào handler.length để quyết định cách gọi callback.
Lưu ý: Đây là cách sử dụng minh họa trong bài học, không phải quy tắc bắt buộc khi thiết kế callback hiện đại.
Kiểm tra nhanh
<!-- 
const fn1 = () => {};
const fn2 = (a, b) => {};
const fn3 = (a, b = 10) => {};

console.log(fn1.length);
console.log(fn2.length);
console.log(fn3.length);
 -->

4. Custom properties
4.1. Thêm properties cho function
Function là 1 object đặc biệt có thể thêm propeties cho nó
<!-- 
function sayHi() {
    console.log("Hello");
}

sayHi.counter = 0;

sayHi.counter++;

console.log(sayHi.counter); // 1
 -->
4.2. Property không phải local variable
<!-- 
function sayHi() {
    console.log("Hello");

    sayHi.counter++;
}

sayHi.counter = 0;

sayHi();
sayHi();

console.log(sayHi.counter); // 2
 -->
4.3. Dùng function property để tạo counter
<!-- 
function makeCounter() {
    function counter() {
        return counter.count++;
    }

    counter.count = 0;

    return counter;
}

let counter = makeCounter();

console.log(counter()); // 0
console.log(counter()); // 1
console.log(counter()); // 2
 -->
 Ở đây:
-couter.count lưu trạng thái bộ đếm
Mỗi lần gọi counter() , giá trị hiện tại được trả về rồi tăng nên 1
Function có thể truy cập property của chính nó thông qua tên counter
4.4. So sánh function property với closure
Cách 1: Closure
<!-- 
function makeCounter() {
    let count = 0;

    return function() {
        return count++;
    };
}
 -->
Cách 2: Function property
<!-- 
function makeCounter() {
    function counter() {
        return counter.count++;
    }

    counter.count = 0;

    return counter;
}
 -->

 | Closure                               | Function property                          |
 | ------------------------------------- | ------------------------------------------ |
 | Lưu `count` trong lexical environment | Lưu `count` trên function object           |
 | Không truy cập trực tiếp từ bên ngoài | Có thể truy cập qua `.count`               |
 | Trạng thái được giữ kín hơn           | Trạng thái có thể bị thay đổi từ bên ngoài |

 Bài tập :
 <!-- 
 function counter() {
    return counter.count++;
}

counter.count = 5;

console.log(counter());5
console.log(counter());6
console.log(counter.count);7 
  -->
Điểm quan trọng ở toán từ counter.count++;
Trả về giá trị hiện tại trước
Sau đó mới tăng giá trị nên 1

5. Named Function Expression (NFE)
Có 2 cách khai báo function:
Function Declaration:
<!-- 
function sayHi() {
    console.log("Hello");
}
 -->
 Function Expression:
 <!-- 
 let sayHi = function() {
    console.log("Hello");
};
  -->

Function Expression có thể được đặt tên:
<!-- 
let sayHi = function greet() {
    console.log("Hello");
};
 -->
 Đây được gọi là Named function Expression (NFE)
 Trong ví dụ trên :
 sayHi là biến bên ngoài lưu function
 greet là tên bên trong của function expresion
 5.2. Tên bên trong NFE có tác dụng gì?
 Tên greet có thể được sử dụng bên trong chính function đó:
 <!-- 
 let sayHi = function greet() {
    console.log("Hello");

    console.log(greet.name);
};

sayHi();
  -->
Nhưng bên ngoài bạn không thể truy cập greet
<!-- 
let sayHi = function greet() {
    console.log("Hello");
};

greet(); // ReferenceError
 -->
Vì greet chỉ tồn tại bên trong function expression đó thôi
5.3. Tại sao NFE hữu ích?
Một ứng dụng quan trọng là fuction có thể tự gọi lại chính nó (recursion) thông qua bên trong
Ví dụ:
<!-- 
let sayHi = function greet(name) {
    if (name) {
        console.log("Hello, " + name);
    } else {
        greet("Martin");
    }
};

sayHi(); // Hello, Martin
 -->
Ở đây, greet("Martin") gọi lại chính function hiện tại.
Ghi nhớ:
-NFE là Function Expression có tên
- Tên bên trong chỉ truy cập được từ bên trong function
- Tên đó hữu ích khi function cần tự gị lai chính nó
Bài tập nhỏ 5.4
<!-- 
let sayHi = function greet() {
    console.log("Hello");
};

console.log(sayHi.name);
console.log(greet.name);
 -->
 5.5. Tại sao không nên chỉ dùng tên biến bên ngoài?
 <!-- 
 let sayHi = function() {
    console.log("Hello");

    sayHi();
};
  -->
Ở đây functio tự gọi lại thông qua biến bên ngoài sayHi()
Điêu gì sảy ra nếu biến sayHi bị thay đổi?
<!-- 
let sayHi = function() {
    console.log("Hello");
    sayHi();
};

let welcome = sayHi;

sayHi = null;

welcome(); 

 -->

let sayHi = function greet() {
    console.log("Hello");
    greet();
};

let welcome = sayHi;

sayHi = null;

welcome(); // Function vẫn gọi được greet()
Điểm quan trọng: greet là tên nội bộ của function , không phụ thuộc vào biến bên ngoài sayHI . Vì vậy , dù sayHi bị thay đổi
 function vẫn có thể tự gọi lại bằng greet
Bài tập:
<!-- 
let first = function inner() {
    console.log("Hi");
    inner();
};

let second = first;
first = null;

second();
 -->

1. second = first: cả hai cùng tham chiếu đến một function.
2. first = null: chỉ thay đổi biến first, không xóa function.
3. second(): vẫn gọi được function.
4. inner(): gọi lại chính function thông qua tên nội bộ.
5. Quá trình lặp vô hạn cho đến khi tràn call stack.
Bạn đã nắm được:
- Function Expression có tên.
- Phạm vi truy cập của tên nội bộ.
- Cách NFE tự gọi lại chính nó.
- Vì sao NFE an toàn hơn khi biến bên ngoài bị thay đổi.
